async def _filter_already_succeeded_targets(
    self, targets: list[str]
) -> tuple[list[str], list[str]]:
    """Return (targets_to_publish, skipped_targets) using prior same-day history."""
    if self._db is None or self._publish_history is None:
        return targets, []
    to_publish: list[str] = []
    skipped: list[str] = []
    for target in targets:
        records = await self._publish_history.query(
            self._db, publisher_id=target, digest_date=self._run_date, limit=10
        )
        if any(r.status == "success" for r in records):
            skipped.append(target)
        else:
            to_publish.append(target)
    return to_publish, skipped
```

CURATE_SCORE_MIN = 1.0
CURATE_SCORE_MAX = 10.0
CURATE_SUMMARY_MAX_CHARS = 100


def _score_value(value: object) -> float | None:
    """Read a numeric LLM score in [1, 10]; reject out-of-range or non-numeric."""
    if value is None:
        return None
    if isinstance(value, str):
        try:
            value = float(value.strip())
        except ValueError as exc:
            raise WorkflowError("curation record requires numeric score") from exc
    if not isinstance(value, int | float) or isinstance(value, bool):
        raise WorkflowError("curation record requires numeric score")
    score = float(value)
    if not (CURATE_SCORE_MIN <= score <= CURATE_SCORE_MAX):
        raise WorkflowError(
            f"curation score {score} out of range "
            f"[{CURATE_SCORE_MIN}, {CURATE_SCORE_MAX}]"
        )
    return score
```

## 八、风险与取舍

1. **粗粒度幂等的语义**：`(publisher_id, digest_date)` 意味着"同一天同一渠道只发一次"。若精选内容变了（如 LLM 重跑结果不同），仍会跳过——这是预期行为（避免扰民），符合用户决策。
2. **title 回绑的代价**：CURATE_PROMPT 要求 LLM 翻译/改写标题为中文，回绑后该能力失效，中文日报标题回到原始 RSS 标题（可能含英文）。**安全优先于便利**，由 summary 承担中文表达。如果后续产品坚持要 LLM 改写标题，可改为"白名单净化 + 长度上限"方案单独排期。
3. **`_ingest` 全失败的"items 为空"判断**：依赖 `_recent_daily_candidates` 的 7 天 `_CONTENT_FALLBACK_DAYS` 窗口。若历史窗口内有数据，全失败仍会继续用旧库存（向后兼容）；只有完全无候选才 raise。这是产品决策（用旧库存总比不发好），通过 result 暴露 `fetched_counts` 让调用方知晓。
4. **score_reason 渲染**：飞书/企微渲染器不消费该字段，仅作数据留存，未来 P40 成本看板或 P45 反馈闭环可读。
5. **preview 路径也要过滤**：T1 预检门同时作用于 `preview_first` 和 final 两条路径，避免预览重发。
